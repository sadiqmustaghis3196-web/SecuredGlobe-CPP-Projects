cmake_minimum_required(VERSION 3.16)
project(SecuredGlobe VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Sql Network PrintSupport Test)
file(GLOB_RECURSE BANK_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/models/*.cpp src/repositories/*.cpp src/services/*.cpp src/utils/*.cpp)
add_library(banking STATIC ${BANK_SOURCES} resources.qrc)
target_include_directories(banking PUBLIC src)
target_link_libraries(banking PUBLIC Qt6::Core Qt6::Sql Qt6::Network)
file(GLOB_RECURSE UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp)
add_library(banking_ui STATIC ${UI_SOURCES})
target_link_libraries(banking_ui PUBLIC banking Qt6::Widgets Qt6::PrintSupport)
add_executable(SecuredGlobe src/main.cpp)
target_link_libraries(SecuredGlobe PRIVATE banking_ui)
if(MSVC)
 target_compile_options(banking PRIVATE /W4 /permissive-)
 target_compile_options(SecuredGlobe PRIVATE /W4 /permissive-)
else()
 target_compile_options(banking PRIVATE -Wall -Wextra -Wpedantic)
 target_compile_options(SecuredGlobe PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(banking_tests tests/BankingTests.cpp)
 target_link_libraries(banking_tests PRIVATE banking Qt6::Test)
 add_test(NAME banking_services COMMAND banking_tests)
 add_executable(gui_tests tests/GuiTests.cpp)
 target_link_libraries(gui_tests PRIVATE banking_ui Qt6::Test)
 add_test(NAME gui_workflows COMMAND gui_tests)
 set_tests_properties(gui_workflows PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
 add_test(NAME gui_smoke COMMAND SecuredGlobe --smoke-test)
 set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
endif()
install(TARGETS SecuredGlobe RUNTIME DESTINATION bin)
